% Limiti globali: nessuna immagine più larga della riga o più alta del 65% del
% testo. \maxwidth e \maxheight sono definiti qui (non dipendono dalla versione
% di Pandoc) e usati come chiavi predefinite di graphicx. Le chiavi "max width"
% e "max height" di adjustbox impostate con \setkeys{Gin} NON vanno usate: con
% Pandoc < 3.6 annullano il ridimensionamento e le immagini escono dalla pagina.
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>0.65\textheight 0.65\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}
