# 1. Generate a random 256-bit master key (64 hex characters) ONCE (do NOT share this value
#    with anyone, and do not reuse example keys from documentation).
export SPACEGL_KEY="$(openssl rand -hex 32)"

# 2. Start the server (terminal 1).
./spacegl-server

# 3. Start the client (terminal 2) using THE SAME key value.
#
# With GLFW 3d:
export SPACEGL_KEY="<paste the same value generated above>"
./spacegl-client gl

# With Vulkan 3d:
export SPACEGL_KEY="<paste the same value generated above>"
./spacegl-client vk

# NOTE: server and clients MUST run with the identical SPACEGL_KEY,
# otherwise the secure handshake will be rejected.
# The launchers take the key from SPACEGL_KEY_FILE, SPACEGL_KEY or
# ~/.spacegl/key, in this order; only when none exists they ask for it on
# the terminal (not echoed). For a permanent setup prefer a key file:
#   export SPACEGL_KEY_FILE=~/.spacegl/master.key   (64 hex digits, chmod 600)
# Launcher options: --no-banner (no banner, no screen clearing); see
# spacegl-server(1), spacegl-client(1), spacegl-diag(1).

# 4. Server identity (wire revision 4). On its first boot the server creates
#    its own identity key, server_identity.key in its data directory, and
#    prints its fingerprint ("Server identity ...: SHA256:..."). BACK IT UP
#    together with the data directory.
#    On the first connection the client prints the same fingerprint and
#    remembers it in ~/.spacegl/known_servers: check that the two match.
#    If a server later presents another identity the client refuses it
#    ("SERVER IDENTITY CHANGED"); if the change is legitimate (the server's
#    data directory was replaced), remove that server's line from
#    ~/.spacegl/known_servers.
#    The connection itself runs inside TLS 1.3 (wire revision 5) whose
#    certificate is this identity: nothing to install, nothing to renew.
#    Update the server and every client together: an older client is
#    refused at the first byte.
#    To pin the key without trusting the first connection:
./spacegl-server --print-identity        # fingerprint + public key (64 hex)
export SPACEGL_SERVER_KEY="<the 64 hex digits printed above>"

# 5. Vulkan viewer: the GPU-driven renderer is the default (it falls back to
#    the CPU-driven one by itself); to force the CPU-driven renderer:
#    SPACEGL_GPD=0 ./spacegl-client vk
